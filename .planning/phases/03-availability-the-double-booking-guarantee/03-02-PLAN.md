    export const BOOKING_HORIZON_DAYS = 90;   // D-26 platform default
    export type Slot = { startUtc: string; endUtc: string };
    // Enumerate on-the-hour 60-min slots for a venue-local date + wall-clock [openHour, closeHour) window.
    // month is 0-based (JS Date convention). Build every instant with TZDate — never add fixed ms.
    export function slotsForWindow(y: number, m: number, d: number, openHour: number, closeHour: number, tz: string): Slot[] {
      const slots: Slot[] = [];
      for (let h = openHour; h < closeHour; h++) {
        const start = new TZDate(y, m, d, h, 0, 0, tz);
        const end   = new TZDate(y, m, d, h + 1, 0, 0, tz);
        slots.push({ startUtc: start.toISOString(), endUtc: end.toISOString() });
      }
      return slots;
    }
    // Venue-local day-of-week (0=Sun..6=Sat) for operating_hours lookup — computed in the venue tz.
    export function venueDayOfWeek(y: number, m: number, d: number, tz: string): number {
      return new TZDate(y, m, d, 12, 0, 0, tz).getDay();   // noon avoids any midnight-boundary ambiguity
    }
    export function slotStartsInFuture(startUtc: string, now: Date): boolean { return new Date(startUtc).getTime() > now.getTime(); }
    export function isWithinHorizon(startUtc: string, now: Date, horizonDays = BOOKING_HORIZON_DAYS): boolean {
      const limit = now.getTime() + horizonDays * 24 * 60 * 60 * 1000;
      return new Date(startUtc).getTime() <= limit;
    }

    Create src/lib/availability/units.ts. The correctness rests on the CONSTRAINT, not the SELECT (RESEARCH Pattern 2 "Key principle") — the find-free SELECT is an optimization; the constraint is the authority. Never do "query then insert" as a guard.
    ```ts
    import { randomUUID } from "node:crypto";
    import { sql } from "drizzle-orm";
    import { booking } from "@/lib/db/schema";
    import { isPgError } from "@/lib/pg";
    export class NoUnitAvailableError extends Error {}
    export async function createBooking(dbConn, { listingId, bookerId, startsAt, endsAt, unitCount, status = "confirmed" }): Promise<number> {
      for (let attempt = 0; attempt < unitCount; attempt++) {
        const occupied = await dbConn.execute(sql`
          SELECT DISTINCT unit FROM booking
          WHERE listing_id = ${listingId} AND status IN ('pending','confirmed')
            AND tstzrange(starts_at, ends_at, '[)') && tstzrange(${startsAt}, ${endsAt}, '[)')`);
        const taken = new Set((occupied as unknown as { unit: number }[]).map(r => r.unit));  // adapt to postgres.js result shape
        let unit: number | null = null;
        for (let u = 1; u <= unitCount; u++) { if (!taken.has(u)) { unit = u; break; } }
        if (unit == null) throw new NoUnitAvailableError();
        try {
          await dbConn.insert(booking).values({ id: randomUUID(), listingId, bookerId, unit, startsAt, endsAt, status });
          return unit;                                   // won the slot
        } catch (e) {
          if (isPgError(e, "23P01")) continue;           // lost the race for THIS unit → recompute + retry
          throw e;
        }
      }
      throw new NoUnitAvailableError();
    }
    // Map a booking failure to a clean user-facing message (SC#4). Both an exhausted-units result and a
    // raw 23P01 (defense in depth) surface the SAME friendly copy — never a raw 500.
    export function mapBookingError(e: unknown): { error: string } {
      if (e instanceof NoUnitAvailableError || isPgError(e, "23P01")) {
        return { error: "That time was just taken. Pick another slot." };
      }
      throw e;   // unknown errors propagate (real 500)
    }
    ```

## Trust Boundaries

| Boundary | Description |
|----------|-------------|
| client-supplied day/time → read model | The booker requests availability for a day; the server derives all state (never trusts client-provided "available" flags). |
| booking insert → DB constraint | createBooking's SELECT is advisory; the EXCLUDE constraint (Plan 01) is the authority. |

## STRIDE Threat Register

| Threat ID | Category | Component | Disposition | Mitigation Plan |
|-----------|----------|-----------|-------------|-----------------|
| T-03-TAMPER-SLOT | Tampering | getAvailability / createBooking | mitigate | Server re-derives availability from operating hours/blocks/occupying bookings + horizon/now gating; slot instants are server-computed via TZDate. The client never supplies what's bookable. |
| T-03-RACE | Tampering / DoS (money-critical) | createBooking insert | mitigate | Correctness rests on the DB EXCLUDE constraint; the find-free SELECT is only an optimization; retry-on-23P01 bounded by unitCount. No app-level conflict guard. |

| T-03-500 | Information Disclosure | error surfacing | mitigate | mapBookingError converts 23P01/exhausted-units to a clean message; unknown errors propagate (not swallowed). |
