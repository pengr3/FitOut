    ```ts
    import { z } from "zod";
    const hhmm = z.string().regex(/^([01]\d|2[0-3]):[0-5]\d$/, "Use 24-hour HH:mm.");
    export const hoursWindowSchema = z.object({
      dayOfWeek: z.number().int().min(0).max(6),
      openTime: hhmm,
      closeTime: hhmm,
    }).refine(w => w.closeTime > w.openTime, { message: "Close time must be after open time.", path: ["closeTime"] });
    export const weeklyHoursSchema = z.object({
      windows: z.array(hoursWindowSchema),
    }).superRefine((val, ctx) => {
      // No two windows on the same weekday may overlap ('[)' semantics: touching endpoints are OK).
      const byDay = new Map<number, {o:string;c:string}[]>();
      for (const w of val.windows) { (byDay.get(w.dayOfWeek) ?? byDay.set(w.dayOfWeek, []).get(w.dayOfWeek)!).push({o:w.openTime,c:w.closeTime}); }
      for (const list of byDay.values()) {
        list.sort((a,b)=>a.o.localeCompare(b.o));
        for (let i=1;i<list.length;i++) if (list[i].o < list[i-1].c)
          ctx.addIssue({ code: z.ZodIssueCode.custom, message: "These hours overlap. Adjust them so each block stands on its own.", path: ["windows"] });
      }
    });
    export const blockSchema = z.object({
      date: z.string().regex(/^\d{4}-\d{2}-\d{2}$/),
      wholeDay: z.boolean(),
      startTime: hhmm.optional(),
      endTime: hhmm.optional(),
      unit: z.number().int().min(1).nullable(),   // null = whole listing (D-24)
      reason: z.string().max(200).optional(),
    }).superRefine((val, ctx) => {
      if (!val.wholeDay) {
        if (!val.startTime || !val.endTime)
          ctx.addIssue({ code: z.ZodIssueCode.custom, message: "Pick a start and end time to block.", path: ["startTime"] });
        else if (val.endTime <= val.startTime)
          ctx.addIssue({ code: z.ZodIssueCode.custom, message: "Pick an end time that's after the start time.", path: ["endTime"] });
      }
    });
    export type WeeklyHoursInput = z.infer<typeof weeklyHoursSchema>;
    export type BlockInput = z.infer<typeof blockSchema>;
    ```

<threat_model>
## Trust Boundaries

| Boundary | Description |
|----------|-------------|
| client form → server action | Host-submitted hours/blocks cross into the server; times/units must be re-validated and never trusted. |
| host session → another host's listing | An authenticated host could target a listingId they don't own (IDOR). |

## STRIDE Threat Register

| Threat ID | Category | Component | Disposition | Mitigation Plan |
|-----------|----------|-----------|-------------|-----------------|
| T-03-IDOR-HOURS | Elevation of Privilege | saveOperatingHours / addBlock / removeBlock | mitigate | `assertOwnership(listingId, userId)` (listing.hostId === session.userId) inside EVERY action before any write, plus the write is re-scoped `WHERE (id AND listingId owned by user)` — the (host) route group is not sufficient alone (RESEARCH Security V4). |
| T-03-VALIDATE | Tampering | hours/blocks input | mitigate | Shared Zod (weeklyHoursSchema/blockSchema) re-validated server-side with safeParse → fieldErrors; close>open, no-overlap, block-range enforced server-side (RESEARCH Security V5). |
| T-03-BLOCK-UNBLOCK | Tampering | removeBlock | mitigate | Delete scoped to (blockId AND listingId) behind assertOwnership so a non-owner cannot unblock someone's listing. |
| T-03-TZNAIVE | Tampering | addBlock time conversion | mitigate | Block start/end stored as timestamptz (UTC) derived from the venue-local date via TZDate — never naive/local timestamps (CLAUDE.md). |
</threat_model>
