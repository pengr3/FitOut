```typescript
import { format } from "date-fns";
import { tz } from "@date-fns/tz";
import { windowHours } from "@/lib/booking/pricing";

function compose(input: WhenLabelInput, dateFormat: string): string {
  const inTz = tz(input.timezone);
  const hours = windowHours(input.startsAt, input.endsAt);
  const quoted = input.quotedTotalCents ?? 0;
  const hourlyTotal = input.hourlyRateCents != null ? input.hourlyRateCents * hours : null;
  const fullDay = hourlyTotal == null || quoted !== hourlyTotal;
  const dateLabel = format(input.startsAt, dateFormat, { in: inTz });
  const timeLabel = fullDay
    ? "Full day"
    : `${format(input.startsAt, "h:mm a", { in: inTz })} – ${format(input.endsAt, "h:mm a", { in: inTz })}`;
  return `${dateLabel}, ${timeLabel}${input.city ? ` (${input.city} time)` : ""}`;
}

export const composeWhenLabel = (input: WhenLabelInput) => compose(input, "EEEE, MMM d");
export const composeWhenLabelShort = (input: WhenLabelInput) => compose(input, "EEE, MMM d");
```

```typescript
/** D-102: `completed` is DERIVED AT READ TIME and NEVER stored. A `confirmed` booking whose endsAt has
 *  passed displays as Completed — zero cron, zero writes, cannot drift, and cannot race the occupancy
 *  predicate. The `completed` enum value stays unused. `now` is passed IN and must come from the DB clock
 *  at the call site (never Date.now() here) — the same rule the refund preview follows. */
export function deriveDisplayStatus(status: BookingDbStatus, endsAt: Date, now: Date): BookingDisplayStatus {
  if (status === "confirmed" && endsAt.getTime() <= now.getTime()) return "completed";
  return status;
}
```
